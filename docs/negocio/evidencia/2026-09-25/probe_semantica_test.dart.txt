final _log = <String, Object?>{};

LsbCard _card(String g) => LsbCard(
      id: g,
      gloss: g,
      displayText: g,
      iconUrl: '',
      categoryId: 'x',
      subcategoryId: 'x',
      contexts: const [],
      priority: 1,
      suggestedNextCardIds: const [],
      isFrequent: true,
      isEmergency: false,
    );

Future<WidgetRef> _app(WidgetTester tester, String ctxId) async {
  late WidgetRef ref;
  late BuildContext bc;
  await tester.pumpWidget(ProviderScope(
    child: MaterialApp(
      home: Consumer(builder: (context, r, _) {
        ref = r;
        bc = context;
        return const Scaffold(body: SizedBox());
      }),
    ),
  ));
  ref.read(contextProvider.notifier).setContext(contextById(ctxId)!);
  await tester.pump();
  _ctx = bc;
  return ref;
}

late BuildContext _ctx;

Future<void> _tap(WidgetTester tester, WidgetRef ref, String gloss) async {
  // No se espera: el modal queda abierto y el test decide qué pulsar.
  // ignore: unawaited_futures
  elegirGlosa(_ctx, ref, _card(gloss));
  await tester.pumpAndSettle();
}

Future<void> _drain(WidgetTester tester) async {
  await tester.pump(const Duration(seconds: 5));
  await tester.pumpAndSettle();
}

String _text(WidgetRef ref) => _asm.assembleStructured(buildFullDeclarationDraft(ref));

  test('P13 CandidateEngine en las pantallas observadas', () {
    List<String> r(List<String> allow, Set<String> fields, {bool conLista = true, InstitutionProfile? p, dynamic node}) =>
        const CandidateEngine()
            .rank(
              available: [for (final g in allow) _card(g)],
              node: node,
              profile: p ?? InstitutionProfile.unknown,
              need: NeedId.complaints,
              zoneFields: fields,
              zoneAllowlist: conLista ? allow.toSet() : const {},
            )
            .map((c) => c.card.gloss)
            .toList();
    for (final ctx in ['amenaza_digital', 'engano_dinero', 'otro', 'identificacion']) {
      for (final z in contextById(ctx)!.zones) {
        final allow = z.glossAllowlist;
        final con = r(allow, z.answerFields.toSet());
        final sin = r(allow, z.answerFields.toSet(), conLista: false);
        _log['P13_${ctx}_${z.id}'] = {
          'mostradas_hoy': con.length == allow.length ? 'todas (${allow.length})' : con,
          'retiradas_si_no_hubiera_lista_blanca': allow.where((g) => !sin.contains(g)).toList(),
        };
      }
    }
    final graph = DialogueGraph.fromJsonString(
        File('assets/dialogue/dialogue_graph.json').readAsStringSync());
    final nodo = graph.match('¿Le robaron el celular?', mode: CardsFlowPurpose.conversationReply)!.node;
    final perfilPolicia = InstitutionProfile.fromJson(
        (jsonDecode(File('assets/business/institution_profiles.json').readAsStringSync())['perfiles'] as List)
            .cast<Map<String, dynamic>>()
            .firstWhere((p) => p['id'] == 'policia'));
    final allow = contextById('denuncia_robo')!.zoneById('objetos')!.glossAllowlist;
    final cards = [for (final g in allow) _card(g)];
    List<double> puntos(InstitutionProfile p) => const CandidateEngine()
        .rank(available: cards, node: nodo, profile: p, need: NeedId.complaints, zoneAllowlist: allow.toSet())
        .map((c) => c.score)
        .toList();
    List<String> orden(InstitutionProfile p) => const CandidateEngine()
        .rank(available: cards, node: nodo, profile: p, need: NeedId.complaints, zoneAllowlist: allow.toSet())
        .map((c) => c.card.gloss)
        .toList();
    _log['P13_nodo'] = '${nodo.id} slots=${nodo.slots} opciones=${nodo.offerableGlosses}';
    _log['P13_orden_sin_perfil'] = orden(InstitutionProfile.unknown);
    _log['P13_orden_policia'] = orden(perfilPolicia);
    _log['P13_mismo_orden'] = orden(InstitutionProfile.unknown).join() == orden(perfilPolicia).join();
    final a = puntos(InstitutionProfile.unknown), b = puntos(perfilPolicia);
    _log['P13_diferencia_puntos'] = {for (var i = 0; i < a.length; i++) (b[i] - a[i]).toStringAsFixed(3)}.toList();
  });

  test('P17 Orden real de preguntas (CONTINUAR sucesivo) por contexto', () {
    for (final ctx in allSelectableContexts) {
      for (final conPrimera in [false, true]) {
        final c = ProviderContainer();
        c.read(contextProvider.notifier).setContext(ctx);
        final z = c.read(semanticZonesProvider.notifier);
        if (conPrimera) {
          final primera = ctx.zoneById(ctx.entryZoneId)!.glossAllowlist.first;
          z.toggleAnswer(primera);
          c.read(sentenceProvider.notifier).setWords(z.orderedGlosses());
        }
        final orden = <String>[c.read(semanticZonesProvider).activeZoneId!];
        for (var i = 0; i < 20; i++) {
          z.goToNextZone();
          final id = c.read(semanticZonesProvider).activeZoneId!;
          if (id == orden.last) break;
          orden.add(id);
        }
        final nunca = [for (final zz in ctx.zones) if (!orden.contains(zz.id)) zz.id];
        _log['P17_${ctx.id}_${conPrimera ? 'con_' + ctx.zoneById(ctx.entryZoneId)!.glossAllowlist.first : 'sin_respuesta'}'] =
            {'orden': orden, 'nunca_se_visitan': nunca};
        c.dispose();
      }
    }
  });

  });
